      btn.addEventListener('click', () => {

        hideWelcome();
        const target = btn.getAttribute('data-target');
        const panel = document.getElementById(target);

    function hideAllPanels(){
      document.querySelectorAll('.form-panel').forEach(el => el.classList.add('hidden'));
    }

    function hideWelcome(){
      const welcome = document.getElementById('welcome-message');
      if(welcome) welcome.classList.add('hidden');
    }

    // Basic password-strength check
    function isPasswordStrong(pw){
      if(!pw) return false;
      return pw.length >= 8 && /[a-zA-Z]/.test(pw) && /\d/.test(pw);
    }

    // CRITICAL FIX: Generic form submit handler with data harmonization
    async function submitForm(formId, roleLabel){
      const form = document.getElementById(formId);
      const msgEl = document.getElementById(formId.replace('-form', '-msg'));
      if(!form) return;

      if(!form.checkValidity()){
        form.reportValidity();
        if(msgEl) msgEl.textContent = 'Please fix the highlighted fields.';
        return;
      }

      const pwd = form.querySelector('input[type="password"]');
      if(pwd && !isPasswordStrong(pwd.value)){
        const err = form.querySelector('.error');
        if(err) err.style.display = 'block';
        if(msgEl) msgEl.textContent = 'Please choose a stronger password.';
        return;
      } else {
        const err = form.querySelector('.error');
        if(err) err.style.display = 'none';
      }

      // --- START: Data Harmonization to Match Mongoose Schema ---
      const rawData = {};
      Array.from(new FormData(form).entries()).forEach(([k,v]) => { rawData[k] = v; });
      
      const data = {};
      data.role = roleLabel;

      // Map raw data keys (e.g., agency-name, ent-large-budget) to generic schema keys (name, budget)
      for (const key in rawData) {
          let genericKey = key;
          
          // Remove prefixes (e.g., agency-, startup-, ent-large-, ent-small-)
          if (key.startsWith('agency-') || key.startsWith('startup-') || key.startsWith('expert-')) {
              genericKey = key.substring(key.indexOf('-') + 1);
          } else if (key.startsWith('ent-large-') || key.startsWith('ent-small-')) {
              genericKey = key.substring(key.indexOf('-') + 7); // Removes e.g., 'ent-large-'
          }
          
          // Handle specific re-mapping used by the schema
          if (genericKey === 'special') genericKey = 'specialization'; 
          if (genericKey === 'cred') genericKey = 'credibility'; 
          
          // Map to correct final field names:
          const finalKey = {
              'name': 'name', 'contact': 'contact', 'email': 'email', 
              'website': 'website', 'portfolio': 'portfolio', 'service': 'service', 
              'specialization': 'specialization', 'pricing': 'pricing', 'legal': 'legal', 
              'credibility': 'credibility', 'years': 'years', 'team': 'team', 
              'timing': 'timing', 'about': 'about', 'password': 'password',
              'industry': 'industry', 'budget': 'budget', 'data': 'data_size' // CRITICAL: 'data' maps to 'data_size'
          }[genericKey] || genericKey;

          // Trim all string values and save to the final data object
          data[finalKey] = (typeof rawData[key] === 'string') ? rawData[key].trim() : rawData[key];
      }
      // --- END: Data Harmonization ---


      if(msgEl) msgEl.textContent = 'Submitting...';

      try {
        const res = await fetch('http://localhost:5000/api/signup', {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify(data)
        });
        if(!res.ok){
          const json = await res.json();
          if(msgEl) msgEl.textContent = json.message || 'Server error: ' + res.statusText;
          return;
        }
        const json = await res.json();
        if(json.success){
          if(msgEl) msgEl.textContent = 'Registration successful — redirecting...';
          setTimeout(()=> location.href = 'login.html', 1200);
        } else {
          if(msgEl) msgEl.textContent = json.message || 'Registration failed.';
        }
      } catch (err){
        if(msgEl) msgEl.textContent = 'Network error. Please try again later.';
      }
    }

    document.querySelectorAll('input[type=number]').forEach(inp => inp.addEventListener('wheel', (ev)=> ev.target.blur()));
